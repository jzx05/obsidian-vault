# 反转链表
class Solution {

public:

    ListNode* reverseList(ListNode* head) {

        ListNode* dummy = new ListNode();

        ListNode* curr = head;

  

        while(curr) {

            ListNode* next = curr->next;

            curr->next = dummy->next;

            dummy->next = curr;

            curr = next;

        }

        ListNode* result = dummy->next;

        delete dummy;

        return result;

    }

};
